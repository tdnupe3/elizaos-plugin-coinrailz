Tool Implementation (TypeScript):

import { BaseTool } from 'langchain/tools';
import { z } from 'zod';
import { x402Requests } from '@x402/client';

interface CoinRailzInput {
  service: string;
  amount: string;
  reference: string;
}

export class CoinRailzTool extends BaseTool {
  name = 'coinrailz_tool';
  description = 'Pays a Coin Railz API endpoint (with x402) and returns the response.';
  schema = z.object({
    service: z.string().describe('The name or endpoint of the Coin Railz service'),
    amount: z.string().describe('Amount in USDC to pay'),
    reference: z.string().optional().describe('Payment reference')
  });

  async _call({ service, amount, reference }: CoinRailzInput): Promise<string> {
    // Construct x402 payment
    const client = x402Requests();
    const payment = await client.post(`https://api.coinrailz.com/${service}`, {
      amount,
      recipient: service,  // assuming service is contract address
      reference: reference ?? 'agent-request',
      currency: 'USDC'
    });
    // Return result or acknowledgement
    return `Payment of ${amount} USDC to ${service} done. Response: ${JSON.stringify(payment.data)}`;
  }
}

import React, { useState } from 'react';
import { Modal, Button, Text } from '@onchainkit/ui'; 
import { useWallet } from '@onchainkit/hooks'; // hypothetical hook for wallet
import { x402Requests } from '@x402/client';

type X402PaymentProps = {
  amount: string;            // e.g. "0.75"
  recipient: string;         // service address
  reference?: string;
  onSuccess?: () => void;
  onFailure?: (error: string) => void;
};

export const X402Payment: React.FC<X402PaymentProps> = ({ amount, recipient, reference, onSuccess, onFailure }) => {
  const { address, connect, sendTransaction } = useWallet('base'); // Base mainnet
  const [status, setStatus] = useState<'idle'|'pending'|'done'|'error'>('idle');
  const [error, setError] = useState<string>('');

  const handlePay = async () => {
    if (!address) {
      await connect();
    }
    setStatus('pending');
    try {
      // Create signed payload (EIP-712) via x402
      const client = x402Requests({ from: address });
      const payload = await client.createPayment({
        amount,
        recipient,
        reference: reference || 'agent-request',
        currency: 'USDC'
      });
      // Send raw transaction (Base, USDC)
      const txHash = await sendTransaction({
        to: recipient,
        data: payload.to,         // data field from EIP-712 payload
        value: 0,
        gas: payload.gasLimit     // if included
      });
      setStatus('done');
      onSuccess && onSuccess();
    } catch (err: any) {
      setStatus('error');
      setError(err.message);
      onFailure && onFailure(err.message);
    }
  };

  return (
    <Modal isOpen>
      <Modal.Header>Confirm Payment</Modal.Header>
      <Modal.Body>
        <Text>Pay {amount} USDC to {recipient} on Base</Text>
        {status === 'pending' && <Text>Processing payment...</Text>}
        {status === 'error' && <Text color="red">Error: {error}</Text>}
      </Modal.Body>
      <Modal.Footer>
        <Button onClick={handlePay} disabled={status === 'pending'}>Pay Now</Button>
      </Modal.Footer>
    </Modal>
  );
};

class InferenceService {
  constructor(runtime) {
    this.name = 'coinrailz_inference';
    // Register with Virtuals GAME engine
    runtime.registerAction(this.name, this.handler.bind(this));
  }
  async handler({ input, agentContext }) {
    // 1. Attempt inference without payment
    const result = await this.callLLM(input);
    if (result.requiresPayment) {
      // 2. Return x402 payment request details to client
      return { httpStatus: 402, paymentRequest: { amount: result.cost, recipient: agentContext.agentAddress, reference: 'inference' } };
    }
    // 3. If no payment needed, return result
    return { status: 200, data: result.data };
  }
}

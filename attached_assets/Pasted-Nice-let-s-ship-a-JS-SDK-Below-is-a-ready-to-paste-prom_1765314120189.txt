4. Implement CoinRailzClient Core (src/client.ts)

Create src/client.ts implementing:

import { CoinRailzClientConfig, X402ServiceRequest, X402ServiceResponse, GasPriceOracleResponse, TokenMetadataResponse, PredictionMarketOddsResponse, AgentWalletProvisionResponse } from "./types";

export class CoinRailzClient {
  private apiKey: string;
  private baseUrl: string;
  private timeoutMs: number;

  constructor(config: CoinRailzClientConfig) {
    if (!config.apiKey) {
      throw new Error("CoinRailzClient: apiKey is required");
    }
    this.apiKey = config.apiKey;
    this.baseUrl = config.baseUrl ?? "https://coinrailz.com";
    this.timeoutMs = config.timeoutMs ?? 10_000;
  }

  private async request<T = unknown>(service: string, payload?: unknown): Promise<X402ServiceResponse<T>> {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), this.timeoutMs);

    try {
      const res = await fetch(`${this.baseUrl}/x402/${service}`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "X-API-KEY": this.apiKey,
        },
        body: payload ? JSON.stringify(payload) : "{}",
        signal: controller.signal,
      });

      const status = res.status;
      let json: any = null;
      try {
        json = await res.json();
      } catch {
        // ignore json parse errors, keep json = null
      }

      if (!res.ok) {
        return {
          success: false,
          status,
          raw: json ?? res.statusText,
        };
      }

      return {
        success: true,
        status,
        data: json as T,
        raw: json,
      };
    } finally {
      clearTimeout(timeout);
    }
  }

  // Generic call
  public async call<T = unknown>(service: string, payload?: unknown): Promise<X402ServiceResponse<T>> {
    return this.request<T>(service, payload);
  }

  // Convenience helpers (examples below, more to add)

  public async gasPriceOracle(params?: { chain?: string }): Promise<X402ServiceResponse<GasPriceOracleResponse>> {
    return this.request<GasPriceOracleResponse>("gas-price-oracle", params);
  }

  public async tokenMetadata(params: { chain: string; address: string }): Promise<X402ServiceResponse<TokenMetadataResponse>> {
    return this.request<TokenMetadataResponse>("token-metadata", params);
  }

  public async predictionMarketOdds(params: { marketSlug: string }): Promise<X402ServiceResponse<PredictionMarketOddsResponse>> {
    return this.request<PredictionMarketOddsResponse>("prediction-market-odds", params);
  }

  public async createAgentWallet(params?: { label?: string; metadata?: Record<string, unknown> }): Promise<X402ServiceResponse<AgentWalletProvisionResponse>> {
    return this.request<AgentWalletProvisionResponse>("agent-create-wallet", params);
  }
}


Important:

Use POST, not GET (your docs and first-call-free logic are POST-based).

Use X-API-KEY header by default (align with your current server).

Make sure this works with Node 18 global fetch. If needed, add a tiny isomorphic-fetch or undici polyfill – but only if absolutely required.

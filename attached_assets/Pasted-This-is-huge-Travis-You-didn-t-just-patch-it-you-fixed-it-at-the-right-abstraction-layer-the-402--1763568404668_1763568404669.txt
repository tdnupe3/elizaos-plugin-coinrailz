✅ What you just fixed (and why it’s correct)
1. Localhost override at the 402 layer

x402-express was ignoring your configured resource and deriving it from the incoming request (Host header → localhost in your environment). Your interceptor:

router.use((req, res, next) => {
  const originalEnd = res.end.bind(res);
  res.end = function(chunk, encoding, callback) {
    if (res.statusCode === 402 && chunk) {
      const body = JSON.parse(chunk);
      if (body?.x402Version && body?.accepts) {
        const isProduction = process.env.REPLIT_DEPLOYMENT === '1';
        let publicBaseUrl = isProduction 
          ? 'https://coinrailz.com'
          : 'https://workspace.traviskellogg1.repl.co';
        
        body.accepts = body.accepts.map(req => ({
          ...req,
          resource: req.resource.replace(/http:\/\/localhost:\d+/, publicBaseUrl),
          discoverable: true
        }));
        
        return originalEnd.call(this, JSON.stringify(body), encoding, callback);
      }
    }
    return originalEnd.call(this, chunk, encoding, callback);
  };
  next();
});


This is exactly right because:

You only touch 402 responses (x402 payment requirements).

You only touch responses that look like real x402 bodies (x402Version + accepts).

You replace only the http://localhost:PORT prefix and leave the path intact.

You force discoverable: true on every accepts item.

That’s exactly what the Bazaar crawler needs.

const axios = require("axios");

const BASE = "https://coinrailz.com";

const wellknown = [
  "/.well-known/agent-card.json",
  "/.well-known/x402.json",
  "/.well-known/payment-methods.json",
  "/.well-known/pricing.json"
];

const services = [
  "gas-price-oracle",
  "token-metadata",
  "dex-liquidity",
  "approval-manager",
  "token-sentiment",
  "token-price",
  "transaction-builder",
  "whale-alerts",
  "batch-quote",
  "multi-chain-balance",
  "portfolio-tracker",
  "trending-tokens",
  "wallet-risk",
  "trade-signals",
  "contract-quick-scan",
  "instant-agent-wallet",
  "seamless-chain-bridge",
  "verified-agent-identity"
];

async function getJson(url) {
  try {
    const res = await axios.post(url, {}, { timeout: 7000 });
    return { ok: true, status: res.status, data: res.data };
  } catch (e) {
    return { ok: false, error: e.message };
  }
}

(async () => {
  console.log("🔍 Checking .well-known endpoints:\n");
  for (const path of wellknown) {
    const url = `${BASE}${path}`;
    try {
      const res = await axios.get(url, { timeout: 7000 });
      console.log(`✅ ${url} — status=${res.status}`);
    } catch (e) {
      console.log(`❌ ${url} — ERROR: ${e.message}`);
    }
  }

  console.log("\n🔍 Checking x402 services:\n");

  for (const service of services) {
    const url = `${BASE}/x402/${service}`;
    const result = await getJson(url);

    if (!result.ok) {
      console.log(`❌ ${service} — ERROR: ${result.error}`);
      continue;
    }

    const body = result.data;
    const accepts = body?.accepts || [];
    const first = accepts[0];

    const resource = first?.resource;
    const discoverable = first?.discoverable;

    const resourceOk =
      typeof resource === "string" &&
      resource.startsWith("https://coinrailz.com");

    console.log(
      `${resourceOk && discoverable ? "✅" : "⚠️"} ${service} — status=${result.status} — resource=${resource} — discoverable=${discoverable}`
    );
  }

  console.log("\n🏁 Diagnostics complete.\n");
})();


If most lines are ✅ and any ⚠️ only show up for services you intentionally configured differently, you’re good.

private async discoverFromVerifiedTargets(): Promise<DiscoveredAgentRaw[]> {
  const agents: DiscoveredAgentRaw[] = [];
  
  for (const target of VERIFIED_AGENT_TARGETS) {
    try {
      // Try both Google A2A card paths
      const cardPaths = [
        '/.well-known/agent-card.json',
        '/.well-known/agent.json'
      ];
      
      for (const cardPath of cardPaths) {
        const url = `https://${target.domain}${cardPath}`;
        
        const response = await this.safeFetch(url, {
          headers: {
            'Accept': 'application/json',
            'User-Agent': 'CoinRailz-A2A-Platform/1.0'
          }
        }, 10000);
        
        if (response.ok) {
          const agentCard = await this.safeJsonParse(response);
          
          if (agentCard && agentCard.name) {
            agents.push({
              url: target.domain,
              source: 'verified-a2a-registry',
              channels: {
                webhook: agentCard.endpoints?.['message/send'] || url
              },
              wallet: target.wallet,
              capabilities: agentCard.capabilities || {},
              metadata: {
                platform: target.platform,
                verified: target.verified,
                name: agentCard.name,
                description: agentCard.description,
                protocolVersion: agentCard.protocolVersion
              }
            });
            
            console.log(`✅ Discovered real A2A agent: ${agentCard.name}`);
            break; // Found agent card, no need to try other path
          }
        }
      }
    } catch (error) {
      console.log(`⚠️ Could not reach ${target.domain}:`, error.message);
    }
  }
  
  return agents;
}

async discoverAgentsByWalletActivity(): Promise<DiscoveredAgentRaw[]> {
  const agents: DiscoveredAgentRaw[] = [];
  
  // Check known agent wallet addresses for ENS/basename
  for (const walletAddress of AGENT_WALLET_ADDRESSES) {
    try {
      // Resolve ENS name
      const ensName = await this.resolveENS(walletAddress);
      
      if (ensName) {
        // Try to fetch agent card from ENS domain
        const agentCard = await this.fetchAgentCardFromDomain(ensName);
        
        if (agentCard) {
          agents.push({
            url: `https://${ensName}`,
            source: 'blockchain-wallet-discovery',
            wallet: walletAddress,
            channels: {
              xmtp: true,
              push_protocol: true
            },
            capabilities: agentCard.capabilities || {},
            metadata: {
              ensName,
              verified: true,
              discoveryMethod: 'ens-resolution'
            }
          });
        }
      }
    } catch (error) {
      console.log(`Could not resolve wallet ${walletAddress}:`, error.message);
    }
  }
  
  return agents;
}

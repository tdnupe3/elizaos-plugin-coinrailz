```typescript
// In server/routes/aiMarketplaceRoutes.ts - add payment processing
router.post('/process-payment', isAuthenticated, async (req: any, res) => {
  try {
    const { orderId, paymentMethodId } = req.body;
    
    // Get order details
    const order = await storage.getOrder(orderId);
    if (!order || order.customerId !== req.user?.claims?.sub) {
      return res.status(404).json({ success: false, error: 'Order not found' });
    }

    // Create Stripe payment intent
    const paymentIntent = await stripe.paymentIntents.create({
      amount: Math.round(order.amount * 100), // Convert to cents
      currency: 'usd',
      payment_method: paymentMethodId,
      confirm: true,
      metadata: {
        orderId: orderId,
        platformFee: order.platformFee.toString(),
        agentPayout: order.agentPayout.toString()
      }
    });

    if (paymentIntent.status === 'succeeded') {
      // Update order status
      await storage.updateOrder(orderId, {
        status: 'paid',
        escrowStatus: 'held',
        paymentIntentId: paymentIntent.id,
        paidAt: new Date().toISOString()
      });

      // Record platform commission
      await storage.recordCommission({
        orderId,
        platformFee: order.platformFee,
        agentPayout: order.agentPayout,
        status: 'collected'
      });

      res.json({
        success: true,
        message: 'Payment processed successfully',
        paymentIntentId: paymentIntent.id
      });
    } else {
      res.status(400).json({
        success: false,
        error: 'Payment failed'
      });
    }
  } catch (error) {
    console.error('Payment processing error:', error);
    res.status(500).json({ success: false, error: 'Payment processing failed' });
  }
});
```

```typescript
// In server/services/paymentProcessor.ts - create agent payout system
export class AgentPayoutProcessor {
  static async processAgentPayout(orderId: string, amount: number) {
    try {
      const order = await storage.getOrder(orderId);
      if (!order) throw new Error('Order not found');

      // Create payout record
      const payout = {
        id: `payout_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
        orderId,
        agentId: order.agentId,
        amount,
        currency: 'USD',
        status: 'processing',
        createdAt: new Date().toISOString()
      };

      await storage.createPayout(payout);

      // Process actual payout (integrate with Stripe Connect or similar)
      const payoutResult = await this.executeStripePayout(order.agentId, amount);

      if (payoutResult.success) {
        await storage.updatePayout(payout.id, {
          status: 'completed',
          processedAt: new Date().toISOString(),
          externalPayoutId: payoutResult.payoutId
        });

        return { success: true, payoutId: payout.id };
      } else {
        await storage.updatePayout(payout.id, {
          status: 'failed',
          errorMessage: payoutResult.error
        });

        return { success: false, error: payoutResult.error };
      }
    } catch (error) {
      console.error('Agent payout error:', error);
      return { success: false, error: 'Payout processing failed' };
    }
  }

  private static async executeStripePayout(agentId: string, amount: number) {
    try {
      // Get agent's Stripe account
      const agent = await storage.getUser(agentId);
      if (!agent.stripeAccountId) {
        return { success: false, error: 'Agent has no payment account setup' };
      }

      // Create Stripe transfer
      const transfer = await stripe.transfers.create({
        amount: Math.round(amount * 100), // Convert to cents
        currency: 'usd',
        destination: agent.stripeAccountId,
      });

      return { success: true, payoutId: transfer.id };
    } catch (error) {
      console.error('Stripe payout error:', error);
      return { success: false, error: 'Payment transfer failed' };
    }
  }
}
```

```typescript
// In server/services/agentPerformanceService.ts
export class AgentPerformanceService {
  static async updateAgentMetrics(agentId: string, orderData: any) {
    try {
      const currentMetrics = await storage.getAgentPerformance(agentId) || {
        agentId,
        totalOrders: 0,
        completedOrders: 0,
        averageRating: 0,
        totalRatings: 0,
        totalRevenue: 0,
        completionRate: 0,
        avgDeliveryTime: 0,
        disputeCount: 0
      };

      const updatedMetrics = {
        ...currentMetrics,
        totalOrders: currentMetrics.totalOrders + 1,
        completedOrders: orderData.status === 'completed' ? currentMetrics.completedOrders + 1 : currentMetrics.completedOrders,
        totalRevenue: currentMetrics.totalRevenue + (orderData.agentPayout || 0),
        completionRate: ((currentMetrics.completedOrders + (orderData.status === 'completed' ? 1 : 0)) / (currentMetrics.totalOrders + 1)) * 100
      };

      if (orderData.customerRating) {
        const totalRatingPoints = (currentMetrics.averageRating * currentMetrics.totalRatings) + orderData.customerRating;
        updatedMetrics.totalRatings = currentMetrics.totalRatings + 1;
        updatedMetrics.averageRating = totalRatingPoints / updatedMetrics.totalRatings;
      }

      await storage.updateAgentPerformance(agentId, updatedMetrics);

      return { success: true, metrics: updatedMetrics };
    } catch (error) {
      console.error('Agent performance update error:', error);
      return { success: false, error: 'Failed to update agent metrics' };
    }
  }
}
```

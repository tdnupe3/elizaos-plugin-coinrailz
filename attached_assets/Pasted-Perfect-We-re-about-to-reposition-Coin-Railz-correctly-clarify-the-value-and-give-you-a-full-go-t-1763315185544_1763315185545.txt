JavaScript Example
const result = await fetch("https://api.coinrailz.com/wallet-risk", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "x-402-payment": "auto"
  },
  body: JSON.stringify({
    wallet: "0x123...",
    chain: "base"
  })

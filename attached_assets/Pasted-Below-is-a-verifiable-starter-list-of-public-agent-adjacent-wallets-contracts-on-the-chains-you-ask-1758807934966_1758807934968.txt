function csvEscape(v) {
  if (v == null) return '';
  const s = String(v);
  return /[,"\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
}
async function writeCSV(rows) {

}

// -------------- Base / Etherscan v2 holders --------------
async function fetchBaseHolders(contract, limit = MAX_PER_CHAIN) {
  if (!ETHERSCAN_API_KEY) throw new Error('ETHERSCAN_API_KEY is required for Base holder scan');
  const PAGE_SIZE = 100;
  const out = [];
  let page = 1;
  while (out.length < limit) {
    const url =
      `${BASESCAN_URL}?chainid=${BASE_CHAIN_ID}&module=token&action=tokenholderlist` +
      `&contractaddress=${contract}&page=${page}&offset=${PAGE_SIZE}&apikey=${ETHERSCAN_API_KEY}`;
    const res = await fetch(url);
    if (!res.ok) throw new Error(`BaseScan HTTP ${res.status}`);
    const data = await res.json();
    // Etherscan v2 standardizes "result"
    const list = data?.result ?? [];
    if (!Array.isArray(list) || list.length === 0) break;

    for (const item of list) {
      const addr = item?.HolderAddress || item?.address;
      const bal = item?.TokenHolderQuantity || item?.balance;
      if (!addr) continue;
      if (MIN_BALANCE_FILTER && BigInt(bal ?? '0') < BigInt(MIN_BALANCE_FILTER)) continue;
      out.push({ address: addr, balance: bal });
      if (out.length >= limit) break;
    }
    page += 1;
    // polite rate limit: ~5 rps free; sleep a bit
    await sleep(250);
  }
  return out;
}

// -------------- Solana holders --------------
async function fetchSolanaHolders(mintStr, limit = MAX_PER_CHAIN) {
  const conn = new Connection(SOLANA_RPC_URL, 'confirmed');
  const mint = new PublicKey(mintStr);

  // Fast seed (top 20)
  const largest = await conn.getTokenLargestAccounts(mint);
  const seed = (largest?.value ?? []).map((a) => ({ pubkey: a.address.toBase58(), amount: a.uiAmountString }));

  // Wider crawl via getProgramAccounts (SPL Token 2022 / Token Program v1)
  // Filter by mint; decode owners out of account data layout is heavy -> instead use getParsedTokenAccountsByMint (paginated via "before" param)
  const holders = new Map(); // address -> balance
  for (const s of seed) holders.set(s.pubkey, s.amount);

  // Paginate parsed token accounts by mint (collect owners with nonzero balance)
  let before = undefined;
  while (holders.size < limit) {
    const page = await conn.getParsedTokenAccountsByMint(mint, { commitment: 'confirmed' }, before);
    const value = page?.value ?? [];
    if (value.length === 0) break;

    for (const v of value) {
      const info = v?.account?.data?.parsed?.info;
      const owner = info?.owner;
      const amount = info?.tokenAmount?.uiAmountString ?? '0';
      if (owner && Number(amount) > 0) {
        holders.set(owner, amount);
      }
      if (holders.size >= limit) break;
    }
    // cursor: "before" = last pubkey (reverse pagination not always supported -> if not, we break)
    const last = value[value.length - 1]?.pubkey?.toBase58?.();
    if (!last) break;
    before = last;
    await sleep(250);
  }

  return [...holders.entries()].map(([address, balance]) => ({ address, balance }));
}

// -------------- XRPL trustlines --------------
async function fetchXrplTrustlines(issuer, currency, limit = MAX_PER_CHAIN) {
  const client = new xrpl.Client(XRPL_WS_URL);
  await client.connect();
  const out = [];
  let marker = undefined;

  while (out.length < limit) {
    const resp = await client.request({
      command: 'account_lines',
      account: issuer,
      limit: 400,
      ...(marker ? { marker } : {}),
    });
    const lines = resp?.result?.lines ?? [];
    for (const l of lines) {
      // Match the currency (e.g., 'XRT') and positive balance (holder)
      if (currency && l.currency !== currency) continue;
      const holder = l.account;
      const bal = l.balance;
      if (Number(bal) > 0) {
        out.push({ address: holder, balance: bal });
        if (out.length >= limit) break;
      }
    }
    marker = resp?.result?.marker;
    if (!marker) break;
    await sleep(250);
  }

  await client.disconnect();
  return out;
}

async function checkXmtpReachability(client, evmAddresses) {
  const results = new Map();
  // XMTP has batch `canMessage()` for arrays in new SDKs; if not, loop one-by-one.
  for (const addr of evmAddresses) {
    try {
      const ok = await client.canMessage(addr);
      results.set(addr, !!ok);
    } catch {
      results.set(addr, false);
    }
    await sleep(50);
  }
  return results;
}

  } catch {
    return false;
  }
}

// -------------- XRPL memo send (optional; be cautious!) --------------
async function xrplSendMemo(to, memoText, drops = '1') {
  if (DRY_RUN || !XRPL_SECRET) return false;
  const client = new xrpl.Client(XRPL_WS_URL);
  await client.connect();
  const wallet = xrpl.Wallet.fromSeed(XRPL_SECRET);
  try {
    const prepared = await client.autofill({
      TransactionType: 'Payment',
      Account: wallet.address,
      Destination: to,
      Amount: drops, // 1 drop (0.000001 XRP)
      Memos: [{ Memo: { MemoData: Buffer.from(memoText, 'utf8').toString('hex') } }],
    });
    const signed = wallet.sign(prepared);
    const res = await client.submitAndWait(signed.tx_blob);
    await client.disconnect();
    return res?.result?.meta?.TransactionResult === 'tesSUCCESS';
  } catch (e) {
    await client.disconnect();
    return false;
  }
}

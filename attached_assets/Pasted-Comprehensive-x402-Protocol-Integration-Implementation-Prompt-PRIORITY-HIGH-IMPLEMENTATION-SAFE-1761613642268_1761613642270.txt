export class X402PaymentService {
  private readonly TIMEOUT_MS = 30000; // 30 second timeout
  private readonly MAX_RETRIES = 3;
  
  async createPaymentRequest(params: X402PaymentParams): Promise<X402PaymentResult> {
    try {
      // Validate inputs first
      this.validatePaymentParams(params);
      
      // Use circuit breaker pattern
      if (!await this.isX402Available()) {
        throw new Error('x402 service unavailable - fallback to traditional payment');
      }
      
      // Create payment with timeout
      const result = await this.executeWithTimeout(
        this.callX402API(params),
        this.TIMEOUT_MS
      );
      
      // Log success
      console.log('✅ x402 payment created:', result.paymentId);
      return result;
      
    } catch (error) {
      console.error('❌ x402 payment creation failed:', error);
      // DO NOT throw - return graceful error
      return { success: false, error: error.message };
    }
  }
  
  private async isX402Available(): Promise<boolean> {
    // Health check before attempting x402 operations
    try {
      const response = await fetch('https://x402.io/health', { 
        signal: AbortSignal.timeout(5000) 
      });
      return response.ok;
    } catch {
      return false;
    }
  }
}

// Example structure
export function X402PaymentButton({ 
  amount, 
  agentId, 
  onSuccess, 
  onError 
}: X402PaymentProps) {
  const [isProcessing, setIsProcessing] = useState(false);
  const [error, setError] = useState<string | null>(null);
  
  const handleX402Payment = async () => {
    setIsProcessing(true);
    setError(null);
    
    try {
      // Call backend API with timeout
      const response = await fetch('/api/x402/create-payment', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ amount, agentId }),
        signal: AbortSignal.timeout(30000) // 30 second timeout
      });
      
      if (!response.ok) {
        throw new Error('x402 payment failed');
      }
      
      const result = await response.json();
      
      if (result.success) {
        onSuccess(result);
      } else {
        throw new Error(result.error || 'Payment failed');
      }
      
    } catch (error) {
      console.error('x402 payment error:', error);
      setError('Payment failed. Please try another payment method.');
      onError(error);
    } finally {
      setIsProcessing(false);
    }
  };
  
  return (
    <div className="x402-payment-container">
      <Button 
        onClick={handleX402Payment}
        disabled={isProcessing}
        className="bg-blue-600 hover:bg-blue-700"
      >
        {isProcessing ? 'Processing...' : `Pay ${amount} USDC via x402`}
      </Button>
      {error && (
        <div className="text-red-600 text-sm mt-2">{error}</div>
      )}
    </div>
  );
}

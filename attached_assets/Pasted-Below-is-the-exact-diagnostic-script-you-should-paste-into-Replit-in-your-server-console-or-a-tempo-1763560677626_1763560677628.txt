const axios = require("axios");

const BASE = "https://coinrailz.com";

// 1. Check .well-known endpoints first
const wellknown = [
  "/.well-known/agent-card.json",
  "/.well-known/service-manifest.json",
  "/.well-known/payment-methods.json",
  "/.well-known/pricing.json"
];

// 2. All 18 x402 services (add or remove as needed)
const services = [
  "gas-price-oracle",
  "token-metadata",
  "dex-liquidity",
  "approval-manager",
  "token-sentiment",
  "token-price",
  "transaction-builder",
  "whale-alerts",
  "batch-quote",
  "multi-chain-balance",
  "portfolio-tracker",
  "trending-tokens",
  "wallet-risk",
  "trade-signals",
  "contract-quick-scan",
  "instant-agent-wallet",
  "seamless-chain-bridge",
  "verified-agent-identity"
];

async function checkUrl(url) {
  try {
    const res = await axios.get(url, { timeout: 5000 });
    return { status: res.status, ok: true, data: res.data };
  } catch (e) {
    return { ok: false, error: e.message };
  }
}

(async () => {
  console.log("🔍 Checking .well-known endpoints...\n");

  for (const path of wellknown) {
    const url = `${BASE}${path}`;
    const check = await checkUrl(url);
    console.log(url, check.ok ? "✅ OK" : "❌ ERROR", check.ok ? "" : check.error);
  }

  console.log("\n🔍 Checking x402 services...\n");

  for (const service of services) {
    const url = `${BASE}/x402/${service}`;
    const check = await checkUrl(url);

    if (!check.ok) {
      console.log(`❌ ${service} — ERROR: ${check.error}`);
      continue;
    }

    // resource field check
    const resource = check.data.resource;
    const isPublic = resource?.startsWith("https://coinrailz.com");

    console.log(
      `${isPublic ? "✅" : "⚠️"} ${service} — status=${check.status} — resource=${resource}`
    );
  }

  console.log("\n🏁 Done.\n");
})();
